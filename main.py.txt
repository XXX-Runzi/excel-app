    <div class="box">
        <h2>📑 龙兴公馆合同台账收息管理软件</h2>
        <input type="file" id="excelFile" accept=".xlsx, .xls" />
        <button onclick="exportFile()" class="btn" style="background: #007bff; float: right;">📥 导出修改后的Excel表格</button>
        <br><br>
        <table id="dataTable">
            <thead>
                <tr>
                    <th>行号</th>
                    <th>借款日期(D)</th>
                    <th>月应收息费(L)</th>
                    <th>已收息月数(M)</th>
                    <th>下个到期日(O)</th>
                    <th>操作</th>
                </tr>
            </thead>
            <tbody></tbody>
        </table>

    <script>
        let wb, ws, sheetData = [];

        document.getElementById('excelFile').addEventListener('change', function(e) {
            let file = e.target.files[0];
            let reader = new FileReader();
            reader.onload = function(e) {
                let data = new Uint8Array(e.target.result);
                wb = XLSX.read(data, {type: 'array'});
                ws = wb.Sheets[wb.SheetNames[0]];
                sheetData = XLSX.utils.sheet_to_json(ws, {header: 1});
                renderTable();
            };
            reader.readAsArrayBuffer(file);
        });

        function renderTable() {
            let tbody = document.querySelector('#dataTable tbody');
            tbody.innerHTML = '';
            for (let r = 3; r < sheetData.length; r++) {
                let row = sheetData[r];
                if (!row || row.length === 0) continue;
                let tr = document.createElement('tr');
                tr.innerHTML = `
                    <td>${r + 1}</td>
                    <td>${row[3] || ''}</td>
                    <td>${row[11] || 0}</td>
                    <td id="m-${r}">${row[12] || 0}</td>
                    <td id="o-${r}">${row[14] || row[3] || ''}</td>
                    <td><button class="btn" onclick="confirmPayment(${r})">确认收息 (+1月)</button></td>
                `;
                tbody.appendChild(tr);
            }
        }

        function confirmPayment(r) {
            let currentM = parseInt(sheetData[r][12] || 0) + 1;
            sheetData[r][12] = currentM;
            document.getElementById(`m-${r}`).innerText = currentM;
            
            // 顺延到期日一个月
            let baseDate = new Date(sheetData[r][14] || sheetData[r][3] || new Date());
            baseDate.setMonth(baseDate.getMonth() + 1);
            let nextDate = baseDate.toISOString().split('T')[0];
            sheetData[r][14] = nextDate;
            document.getElementById(`o-${r}`).innerText = nextDate;

            alert(`第 ${r + 1} 行合同确认收息成功！已收息月数加1。`);
        }

        function exportFile() {
            let newWs = XLSX.utils.aoa_to_sheet(sheetData);
            wb.Sheets[wb.SheetNames[0]] = newWs;
            XLSX.writeFile(wb, "更新后合同台账.xlsx");
        }
    </script>
